Enhance 符文 · ID 30071–30073 · 链接上一发法术，路径伤害 · ECS 主轨 / Mono 旧轨

1. 是什么

插在主动法术前。施法时在当前法术与上一发仍存活法术（_lastSpellEntity）间拉链，打路径上的单位。

文案：链接上一个施放的法术并对路径造成 int1 点伤害。

CanShootSpellUtils.cs伤害求和
public static int GetLightningChainDamage_FinalPlayerValue(SpellShootData shootData, Wand wand)
{
    SpellConfig[] array = shootData.EnhanceList.Select(e => e.GetFinalConfig()).ToArray();
    float num = 0f;
    foreach (var item in array.Where(e => e.abilityType == SpellAbilityType.LightningChain))
        num += GetDamage_FinalPlayerValue(item.int1, array, item, wand).Result;
    return Mathf.CeilToInt(num);
}
CanShootSpellUtils.csstopTarget 截断
public static RatioValue GetDamage(float baseDamage, IEnumerable<SpellConfig> enhances, SpellConfig stopTarget)
{
    RatioValue ratioValue = new RatioValue(baseDamage);
    foreach (SpellConfig enhance in enhances)
    {
        if (enhance != stopTarget)
        {
            switch (enhance.abilityType) {
            case EnhanceAttackRatio: case FireCrystal: case OverDrive:
                ratioValue.AddRatio(enhance.float1 / 100f); break;
            case Refraction: ratioValue.MulRatio(enhance.float2 / 100f); break;
            case PowerSavingMode: ratioValue.MulRatio(enhance.float3 / 100f); break;
            }
            continue;
        }
        return ratioValue; // 碰到闪电链自身即停
    }
    return ratioValue;
}
